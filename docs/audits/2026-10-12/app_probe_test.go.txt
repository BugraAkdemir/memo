 fail:=false;s,err:=memory.NewStore(memory.StoreConfig{Dir:t.TempDir(),Dimension:4,EmbeddingFunc:func(context.Context,string)([]float32,error){if fail{return nil,fmt.Errorf("connection refused")};return []float32{1,0,0,0},nil}});if err!=nil{t.Fatal(err)};defer s.Close()

 direct,err:=s.GetPinnedFactsRanked(context.Background(),"different query",5);if err!=nil||len(direct)!=1{t.Fatalf("pinned fallback fixture: %v %v",direct,err)}

}

 dir:=auditData(t);sess:=filepath.Join(dir,"sessions");os.MkdirAll(sess,0700);if err:=os.Symlink(filepath.Join(dir,"missing"),filepath.Join(sess,"broken.json"));err!=nil{t.Fatal(err)}

}
